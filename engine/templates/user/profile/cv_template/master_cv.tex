%% ── CV primitives ───────────────────────────────────────────────

% \cvrow{LEFT COLUMN}{RIGHT COLUMN}
% Generic two-column row. All other commands build on this.
\newcommand{\cvrow}[2]{%
  \noindent
  \begin{minipage}[t]{\leftcol}\raggedright\small #1\end{minipage}%
  \hspace{\colsep}%
  \begin{minipage}[t]{\rightcol}\raggedright #2\end{minipage}%
  \par\vspace{5pt}%
}

% \cvskill{Category label}{bullet \item lines}
% Used for Competencies section.
\newcommand{\cvskill}[2]{%
  \cvrow{\textbf{#1}}{%
    \begin{itemize}[nosep, topsep=1pt, leftmargin=1.1em, itemsep=1.5pt,
                   label=\textbullet]
      #2
    \end{itemize}%
  }%
  \vspace{10pt}%   % gap between competency groups — increase/decrease to taste
}

% \cvjob{dates}{Title}{Organisation line}{body — use cvbullets or leave empty}
% Used for Experience and Education entries.
\newcommand{\cvjob}[4]{%
  \cvrow{\textbf{#1}}{%
    \textbf{#2}\\[-1pt]
    {\small\itshape\color{subtle}#3}%
    #4%
  }%
}

% cvbullets environment — indented bullet list inside a cvjob
\newenvironment{cvbullets}{%
  \begin{itemize}[nosep, topsep=3pt, leftmargin=1.1em, itemsep=2pt,
                 label=\textbullet]
}{%
  \end{itemize}%
}
